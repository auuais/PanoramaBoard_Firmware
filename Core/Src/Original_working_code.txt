void Enable_All_Camera_Power(void) {
    // Port C GPIO outputs (SW_EN signals)
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_8, GPIO_PIN_SET);   // +12V_SW_EN
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_9, GPIO_PIN_SET);   // +12V_SW_EN
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_11, GPIO_PIN_SET);  // +12V_SW_EN
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_12, GPIO_PIN_SET);  // +12V_SW_EN
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_13, GPIO_PIN_SET);  // +12V_SW_EN

    // Port A GPIO outputs (SW_EN signals)
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_8, GPIO_PIN_SET);   // +4V_SW_EN
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_15, GPIO_PIN_SET);  // +4V_SW_EN

    // Port D GPIO outputs (SW_EN signals)
    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_0, GPIO_PIN_SET);   // +4V_SW_EN
    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_1, GPIO_PIN_SET);   // +4V_SW_EN
    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_3, GPIO_PIN_SET);   // +4V_SW_EN
    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_4, GPIO_PIN_SET);   // +4V_SW_EN
    HAL_GPIO_WritePin(GPIOD, GPIO_PIN_5, GPIO_PIN_SET);   // +4V_SW_EN
}

void Send_UART3_Debug(const char *format, ...) {
    char buffer[256];
    va_list args;
    va_start(args, format);
    vsnprintf(buffer, sizeof(buffer), format, args);
    va_end(args);
    HAL_UART_Transmit(&huart3, (uint8_t *)buffer, strlen(buffer), HAL_MAX_DELAY);
}

void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin)
{
    HAL_UART_Transmit(&huart3, (uint8_t *)"Callback called!\r\n", 18, 100);

    if (GPIO_Pin == GPIO_PIN_9) {
        drdy_interrupt = 1;
        HAL_UART_Transmit(&huart3, (uint8_t *)"PB9 callback hit!\r\n", 19, 100);
    } else {
        char msg[30];
        snprintf(msg, sizeof(msg), "Wrong pin: %d\r\n", GPIO_Pin);
        HAL_UART_Transmit(&huart3, (uint8_t *)msg, strlen(msg), 100);
    }
}



uint16_t calc_checksum(uint8_t *buff, uint32_t length) {
    uint16_t sum = 0;
    for (uint32_t i = 0; i < length; i++) sum += (uint16_t)buff[i];
    return sum;
}

void hardware_reset(void) {
    Send_UART3_Debug("Resetting IMU sensor...\r\n");
    // PA0 is reset pin (keep low then high)
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_RESET);
    HAL_Delay(300);
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_0, GPIO_PIN_SET);
    HAL_Delay(300);
    Send_UART3_Debug("IMU sensor reset complete.\r\n");
}

float decode_float(const uint8_t *bytes, int is_big_endian) {
    union {
        float f;
        uint32_t i;
    } u;

    if (is_big_endian) {
        u.i = ((uint32_t)bytes[0] << 24) | ((uint32_t)bytes[1] << 16) |
              ((uint32_t)bytes[2] << 8)  | (uint32_t)bytes[3];
    } else {
        u.i = ((uint32_t)bytes[3] << 24) | ((uint32_t)bytes[2] << 16) |
              ((uint32_t)bytes[1] << 8)  | (uint32_t)bytes[0];
    }
    return u.f;
}

void send_imu_command(uint8_t *tx_packet, uint32_t tx_size, uint32_t rx_size, const char *cmd_name) {
    uint8_t rx_packet[70] = {0};
    uint8_t echo_byte = 0;

    Send_UART3_Debug("Sending IMU %s command\r\n", cmd_name);

    // PB12 is NSS pin
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_RESET);
    HAL_Delay(10);
    HAL_StatusTypeDef spi_status = HAL_SPI_Transmit(&hspi2, tx_packet, tx_size, HAL_MAX_DELAY);
    if (spi_status != HAL_OK) {
        Send_UART3_Debug("IMU SPI Transmit Failed!\r\n");
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_SET);
        return;
    }

    HAL_SPI_Receive(&hspi2, &echo_byte, 1, HAL_MAX_DELAY);
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_SET);

    // Wait for DRDY interrupt
    drdy_interrupt = 0;
    uint32_t timeout = 500;
    uint32_t start_time = HAL_GetTick();
    while (!drdy_interrupt && (HAL_GetTick() - start_time < timeout)) {
        HAL_Delay(1);
    }

    if (drdy_interrupt) {
        Send_UART3_Debug("IMU DRDY received for %s\r\n", cmd_name);
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_RESET);
        HAL_Delay(10);
        HAL_SPI_Receive(&hspi2, rx_packet, rx_size + 1, HAL_MAX_DELAY);
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_12, GPIO_PIN_SET);

        uint8_t *actual_packet = rx_packet + 1;
        if (actual_packet[3] == CMD_TYPE_READ_REG && actual_packet[12] == IIM46234_WHO_AM_I) {
            Send_UART3_Debug("IMU IIM46234 sensor detected\r\n");
        }
    } else {
        Send_UART3_Debug("IMU DRDY timeout for %s\r\n", cmd_name);
    }
}

// ===================== DAY CAMERA (VISCA) FUNCTIONS =====================
void select_day_camera(uint8_t camera_num) {
    if (camera_num > 5) camera_num = 5;
    current_day_camera = camera_num;

    // Set day camera multiplexer control pins (A=PC3, B=PB1, C=PB2)
    HAL_GPIO_WritePin(GPIOC, GPIO_PIN_3, (camera_num & 0x01) ? GPIO_PIN_SET : GPIO_PIN_RESET);  // A
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_1, (camera_num & 0x02) ? GPIO_PIN_SET : GPIO_PIN_RESET);  // B
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_2, (camera_num & 0x04) ? GPIO_PIN_SET : GPIO_PIN_RESET);  // C

    HAL_Delay(10);
    Send_UART3_Debug("Selected day camera %d\r\n", camera_num);
}

void parse_and_print_version_response(uint8_t* data, uint8_t len) {
    if (len < 10) {
        Send_UART3_Debug("Day camera incomplete version response\r\n");
        return;
    }
    uint8_t address = (data[0] & 0x0F);
    uint16_t model_code = (data[2] << 8) | data[3];
    uint8_t fw_major = data[4];
    uint8_t fw_minor = data[5];

    Send_UART3_Debug("Day Camera %d - Model: 0x%04X, FW: %d.%03d\r\n",
                     current_day_camera, model_code, fw_major, fw_minor);
}

void wait_for_visca_ack_and_completion(uint32_t timeout_ms, int parse_version) {
    uint8_t got_ack = 0, got_completion = 0;
    uint32_t start = HAL_GetTick();
    while ((HAL_GetTick() - start) < timeout_ms && !(got_ack && got_completion)) {
        if (visca_rx_ready) {
            if (visca_rx_idx >= 3 && visca_rx_buffer[0] == 0x90 && visca_rx_buffer[1] == 0x41) {
                Send_UART3_Debug("Day Camera %d: ACK received\r\n", current_day_camera);
                got_ack = 1;
            } else if (visca_rx_idx >= 3 && visca_rx_buffer[0] == 0x90 && visca_rx_buffer[1] == 0x50) {
                Send_UART3_Debug("Day Camera %d: Completion received\r\n", current_day_camera);
                got_completion = 1;
                if (parse_version)
                    parse_and_print_version_response(visca_rx_buffer, visca_rx_idx);
            }
            visca_rx_ready = 0;
            visca_rx_idx = 0;
            memset(visca_rx_buffer, 0, sizeof(visca_rx_buffer));
        }
    }
    if (!got_ack && !got_completion) {
        Send_UART3_Debug("Day Camera %d: No response\r\n", current_day_camera);
    }
}

void send_visca_command(uint8_t* cmd, uint8_t cmd_len, int parse_version) {
    visca_rx_idx = 0;
    visca_rx_ready = 0;
    memset(visca_rx_buffer, 0, sizeof(visca_rx_buffer));
    HAL_UART_Receive_IT(&huart2, &visca_rx_byte, 1);
    HAL_UART_Transmit(&huart2, cmd, cmd_len, 1000);
    wait_for_visca_ack_and_completion(1000, parse_version);
}

    if (camera_num > 5) camera_num = 5;

    // Set IR camera multiplexer control pins (A=PC0, B=PC1, C=PC2)

uint8_t ComputeChecksum(uint8_t* message_buffer, uint32_t length) {
    uint8_t sum = 0;
    for(uint32_t i = 0; i < length; i++) sum += message_buffer[i];
    return (uint8_t)((~sum) + 1);
}

void Send_IR_Camera_Command(uint8_t message_id, uint8_t *params, uint32_t param_size) {
    uint8_t command_buffer[32];
    command_buffer[0] = 0x01;
    command_buffer[1] = message_id;
    command_buffer[2] = param_size;
    if (param_size > 0 && params != NULL) memcpy(&command_buffer[3], params, param_size);
    uint32_t body_length = 3 + param_size;
    command_buffer[body_length] = ComputeChecksum(command_buffer, body_length);
    HAL_UART_Transmit(&huart1, command_buffer, body_length + 1, 100);

    Send_UART3_Debug("IR Camera %d CMD Sent: 0x%02X, PayloadLen: %lu\r\n", current_ir_camera, message_id, param_size);
}

bool Validate_IR_Parameter(uint16_t param_id, uint16_t value) {
    switch(param_id) {
        case PARAM_ID_NUC_COMMAND:   return value <= 2;
        case PARAM_ID_ROTATION:      return value <= 3;
        case PARAM_ID_FRAMERATE:     return value <= 8;
        case PARAM_ID_PALETTE:       return value >= 1 && value <= 6;
        case PARAM_ID_POLARITY:      return value <= 1;
        case PARAM_ID_D_ZOOM:        return value <= 12;
        case PARAM_ID_TNR_ENABLE:    return value <= 1;
        case PARAM_ID_POWER_RESET:   return value <= 1;
        default: return false;
    }
}

void Translate_And_Send_IR_Camera_Command(uint16_t param_id, uint16_t value) {
    uint8_t payload[4];
    switch(param_id) {
    case PARAM_ID_POLARITY:
        if (value == 1) {
            Send_IR_Camera_Command(0x29, NULL, 0);
            Send_UART3_Debug("IR Camera %d: Set Polarity WHITE HOT\r\n", current_ir_camera);
        } else {
            Send_IR_Camera_Command(0x28, NULL, 0);
            Send_UART3_Debug("IR Camera %d: Set Polarity BLACK HOT\r\n", current_ir_camera);
        }
        break;
    case PARAM_ID_D_ZOOM:
        payload[0] = 0x00;
        payload[1] = (uint8_t)value;
        Send_IR_Camera_Command(0xA4, payload, 2);
        Send_UART3_Debug("IR Camera %d: Set Digital Zoom: %d\r\n", current_ir_camera, value);
        break;
    case PARAM_ID_PALETTE:
        payload[0] = 0x00;
        payload[1] = value;
        Send_IR_Camera_Command(0x58, payload, 2);
        Send_UART3_Debug("IR Camera %d: Set Palette: %d\r\n", current_ir_camera, value);
        break;
    default:
        Send_UART3_Debug("IR Camera %d: Unknown Param ID 0x%04X\r\n", current_ir_camera, param_id);
        break;
    }
}

void Process_IR_Camera_Received_Data(uint8_t* data, uint16_t length) {
    uint16_t current_pos = 0;
    while (current_pos < length) {
        if (data[current_pos] == 0x01) {
            if (current_pos + 3 < length) {
                uint8_t message_id = data[current_pos + 1];
                uint8_t param_len = data[current_pos + 2];
                uint32_t total_msg_len = 4 + param_len;
                if (current_pos + total_msg_len <= length) {
                    uint8_t* msg_ptr = &data[current_pos];
                    if (msg_ptr[total_msg_len - 1] == ComputeChecksum(msg_ptr, total_msg_len - 1)) {
                        if (message_id == 0x00) {
                            Send_UART3_Debug("IR Camera %d Text: %.*s\r\n", current_ir_camera, param_len, (char*)&msg_ptr[3]);
                        } else if (message_id == 0x02) {
                            Send_UART3_Debug("IR Camera %d ACK received\r\n", current_ir_camera);
                        }
                        current_pos += total_msg_len; continue;
                    }
                }
            }
        }
        current_pos++;
    }
}


int IR_FactoryResetNV(void)
{
    uint8_t cmd[4];
    cmd[0] = 0x01;
    cmd[1] = 0xB3;
    cmd[2] = 0x00;
    cmd[3] = ComputeChecksum(cmd, 3);

    // Arm RX cleanly
    ir_camera_data_received_flag = false;
    ir_camera_received_bytes = 0;
    memset(ir_camera_rx_buffer, 0, sizeof ir_camera_rx_buffer);
    HAL_UARTEx_ReceiveToIdle_IT(&huart1, ir_camera_rx_buffer, sizeof ir_camera_rx_buffer);

    HAL_UART_Transmit(&huart1, cmd, sizeof(cmd), 200);

    uint32_t t0 = HAL_GetTick();
    while (HAL_GetTick() - t0 < 1000) {
        if (!ir_camera_data_received_flag) continue;
        ir_camera_data_received_flag = false;

        for (uint16_t i = 0; i + 5 < ir_camera_received_bytes; i++) {
            if (ir_camera_rx_buffer[i]     == 0x01 &&
                ir_camera_rx_buffer[i + 1] == 0x02 &&   // ACK
                ir_camera_rx_buffer[i + 3] == 0x00 &&
                ir_camera_rx_buffer[i + 4] == 0xB3) {
                return 0; // SUCCESS
            }
        }
    }
    return -1; // FAILED
}


int IR_DisableSuperFrame(void)
{
    uint8_t cmd[8];
    uint16_t param_id = 0x0036;   // Super Frame Enable
    uint16_t value    = 0x0000;   // Disable

    cmd[0] = 0x01;
    cmd[1] = 0xB4;                // Customer Setting Set
    cmd[2] = 0x04;
    cmd[3] = (uint8_t)(param_id >> 8);
    cmd[4] = (uint8_t)(param_id & 0xFF);
    cmd[5] = (uint8_t)(value >> 8);
    cmd[6] = (uint8_t)(value & 0xFF);
    cmd[7] = ComputeChecksum(cmd, 7);

    // Arm RX
    ir_camera_data_received_flag = false;
    ir_camera_received_bytes = 0;
    memset(ir_camera_rx_buffer, 0, sizeof ir_camera_rx_buffer);
    HAL_UARTEx_ReceiveToIdle_IT(&huart1, ir_camera_rx_buffer, sizeof ir_camera_rx_buffer);

    HAL_UART_Transmit(&huart1, cmd, sizeof(cmd), 200);

    // Wait for ACK
    uint32_t t0 = HAL_GetTick();
    while (HAL_GetTick() - t0 < 500) {
        if (!ir_camera_data_received_flag) continue;
        ir_camera_data_received_flag = false;

        for (uint16_t i = 0; i + 5 < ir_camera_received_bytes; i++) {
            if (ir_camera_rx_buffer[i]     == 0x01 &&
                ir_camera_rx_buffer[i + 1] == 0x02 &&  // ACK
                ir_camera_rx_buffer[i + 3] == 0x00 &&
                ir_camera_rx_buffer[i + 4] == 0xB4) {
                return 0; // SUCCESS
            }
        }
    }
    return -1; // FAILED
}



//IR Init Helpers

static void IR_ArmRx(void) {
    ir_camera_data_received_flag = false;
    ir_camera_received_bytes = 0;
    memset(ir_camera_rx_buffer, 0, sizeof ir_camera_rx_buffer);
    HAL_UARTEx_ReceiveToIdle_IT(&huart1, ir_camera_rx_buffer, sizeof ir_camera_rx_buffer);
}

static int IR_WaitPacket(uint32_t timeout_ms) {
    uint32_t t0 = HAL_GetTick();
    while (HAL_GetTick() - t0 < timeout_ms) {
        if (ir_camera_data_received_flag) return 0;
    }
    return -1;
}

/* Parse packets of form:
   [0]=0x01, [1]=msg_id, [2]=len, [3..3+len-1]=payload, [last]=checksum
*/
static int IR_FindPacket(uint8_t wanted_id, uint8_t wanted_len, uint16_t *out_payload_u16) {
    uint16_t n = ir_camera_received_bytes;
    for (uint16_t i = 0; i + 4u <= n; i++) {
        if (ir_camera_rx_buffer[i] != 0x01) continue;

        uint8_t id  = ir_camera_rx_buffer[i+1];
        uint8_t len = ir_camera_rx_buffer[i+2];
        uint16_t total = (uint16_t)(4u + len);
        if (i + total > n) continue;

        uint8_t *p = &ir_camera_rx_buffer[i];
        if (p[total-1] != ComputeChecksum(p, total-1)) continue;

        if (id == wanted_id && (wanted_len == 0xFF || len == wanted_len)) {
            if (out_payload_u16 && len >= 2) {
                *out_payload_u16 = ((uint16_t)p[3] << 8) | p[4]; // big-endian payload
            }
            return 0;
        }
    }
    return -1;
}

static int IR_SetNV(uint16_t id, uint16_t val) {
    uint8_t p[4] = { (uint8_t)(id>>8), (uint8_t)(id&0xFF), (uint8_t)(val>>8), (uint8_t)(val&0xFF) };

    IR_ArmRx();
    Send_IR_Camera_Command(0xB0, p, 4);

    if (IR_WaitPacket(500) != 0) return -1;

    // Accept either ACK(0x02) or specific ACK id if your camera uses another one
    if (IR_FindPacket(0x02, 0xFF, NULL) == 0) return 0;

    // If your camera responds differently, temporarily dump raw buffer:
    Send_UART3_Debug("SetNV no ACK. RX bytes=%u\r\n", ir_camera_received_bytes);
    return -1;
}

static int IR_GetNV(uint16_t id, uint16_t *out) {
    uint8_t p[2] = { (uint8_t)(id>>8), (uint8_t)(id&0xFF) };

    IR_ArmRx();
    Send_IR_Camera_Command(0xB5, p, 2);

    if (IR_WaitPacket(500) != 0) return -1;

    // You assumed VALUE packet id=0x45, len=2.
    // Keep that, but if it fails, print raw and adjust.
    uint16_t v=0;
    if (IR_FindPacket(0x45, 0x02, &v) == 0) { *out = v; return 0; }

    Send_UART3_Debug("GetNV: VALUE not found. RX bytes=%u\r\n", ir_camera_received_bytes);
    return -1;
}

static int IR_SetMux(uint16_t sel) {
    uint8_t p[2] = { sel>>8, sel&0xFF };
    Send_IR_Camera_Command(0xD7, p, 2);               // Digital Video Mux Select
    // ACK to 0xD7 will arrive; no value to read back
    return 0;
}

                        uint8_t* msg_ptr = &ir_camera_rx_buffer[current_pos];

                                handshake_success = true;
                                Send_UART3_Debug("\r\n--- !!!!!!!!!!!!! Handshake successful - Valid ACK received ---\r\n");
                                break;

    Send_UART3_Debug("Verify PDVO: NV4=%u NV5=%u NV7=%u NV76=%u\r\n", nv4,nv5,nv7,nv76);

    // --- 30 fps ---
    IR_SetNV(16,0);       // NV#16 Frame Rate = 1 => 30 Hz
    uint16_t nv16=0; IR_GetNV(16,&nv16);
    Send_UART3_Debug("Verify FrameRate: NV16=%u\r\n", nv16);

void Read_HDC3022_Sensor(float* temp, float* hum) {
    uint8_t sensor_data_buffer[6];
    uint8_t trigger_cmd[] = {0x24, 0x00};
    if (HAL_I2C_Master_Transmit(&hi2c2, (0x44 << 1), trigger_cmd, 2, 1000) != HAL_OK) {
        Send_UART3_Debug("I2C Transmit Error\r\n");
        *temp = -999.0f;
        *hum = -999.0f;
        return;
    }
    HAL_Delay(20);
    if (HAL_I2C_Master_Receive(&hi2c2, (0x44 << 1), sensor_data_buffer, 6, 1000) != HAL_OK) {
        Send_UART3_Debug("I2C Receive Error\r\n");
        *temp = -999.0f;
        *hum = -999.0f;
        return;
    }
    uint16_t raw_temp = (sensor_data_buffer[0] << 8) | sensor_data_buffer[1];
    uint16_t raw_hum = (sensor_data_buffer[3] << 8) | sensor_data_buffer[4];
    *temp = 175.0f * ((float)raw_temp / 65535.0f) - 45.0f;
    *hum = 100.0f * ((float)raw_hum / 65535.0f);
}

void HAL_UART_RxCpltCallback(UART_HandleTypeDef *huart) {
    if (huart->Instance == USART2) {  // Day camera (VISCA)

    }
}

void HAL_UARTEx_RxEventCallback(UART_HandleTypeDef *huart, uint16_t Size) {
    if (huart->Instance == USART1) {  // IR camera
        ir_camera_data_received_flag = true;
        ir_camera_received_bytes = Size;
        HAL_UARTEx_ReceiveToIdle_IT(&huart1, ir_camera_rx_buffer, 128);

        temp_sensor_request_flag = true;
        temp_sensor_received_bytes = Size;
        HAL_UARTEx_ReceiveToIdle_IT(&huart3, temp_sensor_rx_buffer, 128);
    }
}
